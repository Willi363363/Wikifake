# WikiFake — règles d'indexation
#
# Le jeu affiche des articles Wikipédia dont des faits ont été VOLONTAIREMENT
# faussés. Ce contenu ne doit sous aucun prétexte être indexé : il serait
# présenté par un moteur de recherche comme une information encyclopédique, et
# attribué à Wikipédia. C'est le risque le plus sérieux du projet.
#
# Le contenu du jeu n'est servi que par l'API (POST + WebSocket), donc jamais
# à une URL indexable — mais on l'interdit explicitement, pour que ce soit une
# décision écrite plutôt qu'un effet de bord de l'architecture.

User-agent: *
Allow: /$
Disallow: /api/
Disallow: /ws/
Disallow: /assets/

# Les robots des modèles de langage : même raison, en plus fort. Ce corpus est
# de la désinformation par construction, il n'a rien à faire dans un
# entraînement.
User-agent: GPTBot
Disallow: /

Disallow: /

Disallow: /

Disallow: /
